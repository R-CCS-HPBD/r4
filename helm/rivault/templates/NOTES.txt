LiteLLM API Access
=============================================================================

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "rivault.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")

{{- else if contains "LoadBalancer" .Values.service.type }}

  watch kubectl get --namespace {{ .Release.Namespace }} svc {{ include "rivault.fullname" . }}
{{- else if contains "ClusterIP" .Values.service.type }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "rivault.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}

  Then access: http://127.0.0.1:{{ .Values.service.port }}/v1/models
{{- end }}

  API Key: {{ .Values.litellm.masterKey }}
